86static inline bool segment2_intersects(const segment2_t a, const segment2_t b) {
87 winding_t b_start_vs_a = vector2_winding(a.start, a.end, b.start);
88 winding_t b_end_vs_a = vector2_winding(a.start, a.end, b.end);
89 winding_t a_start_vs_b = vector2_winding(b.start, b.end, a.start);
90 winding_t a_end_vs_b = vector2_winding(b.start, b.end, a.end);
91
92 // collinear endpoints only touch if they land within the other segment
93 if (b_start_vs_a == WINDING_COLLINEAR && segment2_contains_bounds(a, b.start)) {
94 return true;
95 }
96 if (b_end_vs_a == WINDING_COLLINEAR && segment2_contains_bounds(a, b.end)) {
97 return true;
98 }
99 if (a_start_vs_b == WINDING_COLLINEAR && segment2_contains_bounds(b, a.start)) {
100 return true;
101 }
102 if (a_end_vs_b == WINDING_COLLINEAR && segment2_contains_bounds(b, a.end)) {
103 return true;
104 }
105
106 // otherwise they cross only if each segment straddles the other
107 return (b_start_vs_a != b_end_vs_a) && (a_start_vs_b != a_end_vs_b);
108}
